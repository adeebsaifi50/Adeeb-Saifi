      Add-Type @"
using System;
using System.Runtime.InteropServices;

public class HotKey {
    [DllImport("user32.dll")]
    public static extern bool RegisterHotKey(IntPtr hWnd, int id, uint fsModifiers, uint vk);

    [DllImport("user32.dll")]
    public static extern bool UnregisterHotKey(IntPtr hWnd, int id);

    [DllImport("user32.dll")]
    public static extern bool PeekMessage(out MSG lpMsg, IntPtr hWnd, uint wMsgFilterMin, uint wMsgFilterMax, uint wRemoveMsg);

    [DllImport("user32.dll")]
    public static extern bool TranslateMessage(ref MSG lpMsg);

    [DllImport("user32.dll")]
    public static extern IntPtr DispatchMessage(ref MSG lpMsg);

    [StructLayout(LayoutKind.Sequential)]
    public struct MSG {
        public IntPtr hwnd;
        public uint message;
        public UIntPtr wParam;
        public IntPtr lParam;
        public uint time;
        public int pt_x;
        public int pt_y;
    }
}
"@

Add-Type -AssemblyName System.Windows.Forms

$START = 1
$STOP  = 2

# F8 = START
if (-not [HotKey]::RegisterHotKey([IntPtr]::Zero, $START, 0, 0x77)) {
    Write-Host "F8 register nahi ho paya."
    exit
}

# F9 = STOP
if (-not [HotKey]::RegisterHotKey([IntPtr]::Zero, $STOP, 0, 0x78)) {
    [HotKey]::UnregisterHotKey([IntPtr]::Zero, $START)
    Write-Host "F9 register nahi ho paya."
    exit
}

$running = $false
$nextWork = [DateTime]::Now

# Initialize the message structure for PeekMessage
$msg = New-Object HotKey+MSG

Write-Host ""
Write-Host "=============================="
Write-Host " Automation Ready"
Write-Host " F8 = START"
Write-Host " F9 = STOP"
Write-Host "=============================="
Write-Host ""

try {
    while ($true) {

        # F8 / F9 check
        while ([HotKey]::PeekMessage(
            [ref]$msg,
            [IntPtr]::Zero,
            0,
            0,
            1
        )) {

            if ($msg.message -eq 0x0312) {

                $id = $msg.wParam.ToUInt32()

                if ($id -eq $START) {
                    $running = $true
                    $nextWork = [DateTime]::Now
                    Write-Host "$(Get-Date -Format 'HH:mm:ss')  STARTED"
                }
                elseif ($id -eq $STOP) {
                    $running = $false
                    Write-Host "$(Get-Date -Format 'HH:mm:ss')  STOPPED"
                }
            }

            [HotKey]::TranslateMessage([ref]$msg)
            [HotKey]::DispatchMessage([ref]$msg)
        }

        if ($running -and [DateTime]::Now -ge $nextWork) {

            # Ctrl + N
            [System.Windows.Forms.SendKeys]::SendWait("^n")

            # 300 milliseconds
            Start-Sleep -Milliseconds 300

            # Space
            [System.Windows.Forms.SendKeys]::SendWait(" ")

            # Next run after 5 seconds
            $nextWork = [DateTime]::Now.AddSeconds(5)
        }

        Start-Sleep -Milliseconds 5
    }
}
finally {
    [HotKey]::UnregisterHotKey([IntPtr]::Zero, $START)
    [HotKey]::UnregisterHotKey([IntPtr]::Zero, $STOP)
    Write-Host "Automation closed."
}

}
